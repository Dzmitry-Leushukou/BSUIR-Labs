Авторизация и сессии
	ensure_core_roles() — создать базовые роли admin, user, manager (идемпотентно).

register_user(p_email, p_password, p_name, p_surname) RETURNS INT — регистрация обычного пользователя (+хеш пароля).

login(p_email, p_password, p_ip) RETURNS TABLE(session_id INT, user_id INT) — проверка пароля, создание сессии.

logout(p_user_id) — завершить активную сессию.

change_password(p_user_id, p_old_password, p_new_password) — смена пароля.

touch_session(p_session_id, p_ip) — продлить/пометить сессию активной (обновить updated_at и IP).

Управление пользователями и ролями (админ)

admin_create_user(p_actor_id, p_email, p_temp_password, p_name, p_surname, p_role_name) RETURNS INT

admin_delete_user(p_actor_id, p_user_id, p_hard BOOLEAN DEFAULT FALSE) — бан или жёсткое удаление.

admin_set_role(p_actor_id, p_user_id, p_role_name)

admin_ban_user(p_actor_id, p_user_id) / admin_unban_user(p_actor_id, p_user_id)

admin_list_users(p_search TEXT DEFAULT NULL) — поиск/листинг (email/имя/фамилия/статус).

get_my_profile(p_user_id) — безопасный профиль для клиента (без хеша пароля).
